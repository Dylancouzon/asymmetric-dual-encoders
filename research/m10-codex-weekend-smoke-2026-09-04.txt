    75	        try:
    76	            r = requests.post(base + "/chat/completions", json=body, timeout=timeout)
    77	            r.raise_for_status()
    78	            blob = r.json()
    79	            out_tok += blob.get("usage", {}).get("completion_tokens", 0)
    80	            ch = blob["choices"][0]
    81	            fin = ch.get("finish_reason")
    82	            text = ch["message"]["content"] or ""
    83	        except Exception as e:
    84	            # A malformed reply is a contract failure like any other: retry once, then drop the
    85	            # seed. It must never take the whole run down (Codex 2026-09-04). But a TRANSPORT
    86	            # failure is not evidence about the prompt (Fable 2026-09-04): it is reported
    87	            # separately so a dead server cannot burn a prompt revision or drop a form.
    88	            last, transport = repr(e)[:200], True
    89	            continue
    90	        transport = False
    91	        got = forms.parse(text, n)
    92	        if got is not None:
    93	            return got, True, retry, out_tok, None, fin
    94	        last = text[:200] + (f"  [finish_reason={fin}]" if fin else "")
    95	    return None, False, True, out_tok, last, ("transport" if transport else fin)

    98	def generate(form, seeds, n=5, base=BASE, workers=32, timeout=600, label="", seen=None):
    99	    """seeds: [(passage_id, passage_text)] -> dict with queries, rates and per-seed counts.
   100	
   101	    `seen`: a caller-owned set of normalized query strings, shared across every call so exact
   102	    deduplication is corpus-wide rather than per-call. Mutated in place.
   103	    """
   104	    t0 = time.time()
   105	    with ThreadPoolExecutor(workers) as ex:
   106	        rows = list(ex.map(lambda s: _one(form, s[1], s[0], n, base, timeout), seeds))
   107	    dt = time.time() - t0
   108	    if seen is None:
   109	        seen = set()
   110	    queries, n_emitted = [], 0
   111	    for (qs, _o, _r, _t, _e, _f), (pid, _p) in zip(rows, seeds):
   112	        if qs is None:
   113	            continue
   114	        for q in qs:
   115	            n_emitted += 1
   116	            k = " ".join(q.split()).lower()
   117	            if k and k not in seen:
   118	                seen.add(k)
   119	                queries.append({"query": q.strip(), "form": form, "seed_id": pid})
   120	    ok = [r for r in rows if r[1]]
   121	    reached = [r for r in rows if r[5] != "transport"]
   122	    out_tok = sum(r[3] for r in rows)
   123	    res = dict(form=form, model=MODEL, revision=REVISION, n_seeds=len(seeds), n_per_seed=n,
   124	               max_tokens=max_tokens_for(form, n),
   125	               contract_ok=len(ok), contract_rate=round(len(ok) / max(len(reached), 1), 4),
   126	               contract_rate_all_seeds=round(len(ok) / max(len(seeds), 1), 4),
   127	               reached=len(reached),
   128	               retried=sum(1 for r in rows if r[2]), seconds=round(dt, 1),
   129	               out_tok=out_tok, out_tok_per_s=round(out_tok / max(dt, 1e-9), 1),
   130	               n_queries=len(queries), exact_dupes=n_emitted - len(queries),
   131	               transport_failures=sum(1 for r in rows if r[5] == "transport"),
   132	               truncated=sum(1 for r in rows if r[5] == "length"),
   133	               finish_reasons={k: sum(1 for r in rows if r[5] == k)
   134	                               for k in {r[5] for r in rows} if k},
   135	               first_failures=[r[4] for r in rows if not r[1]][:3],
   136	               queries=queries)
   137	    if label:
   138	        print(f"{label:26s} contract {res['contract_rate']:.3f}  {res['n_queries']:5d} queries  "
   139	              f"{res['out_tok_per_s']:7.1f} out tok/s  {res['seconds']:6.1f}s", flush=True)
   140	    return res
     1	"""M10 step 1 -- the generation smoke (decision 15).
     2	
     3	Per generated form: 40 seed passages x 5 queries = 200 queries. Two gates:
     4	  * contract >= 90%  -- computed here, from `forms.parse` on the raw reply;
     5	  * on-form  >= 80%  -- a 50-query sample judged by an INDEPENDENT Fable subagent against the
     6	    form's registered description in `forms.FORMS`; this script only produces the sample.
     7	
     8	Outputs `work/m10gen/smoke/<form>.json` (everything) and `m10/SMOKE.md` (the pushable record:
     9	gates, rates, the 50-query judge samples and the prompt hash each form's approval binds to).
    10	"""
    11	import hashlib, json, os, sys, time
    12	from pathlib import Path

    22	N_SEEDS, N_PER_SEED = 40, 5

    25	def prompt_hash(form):
    26	    """Approval binds to this: system turn + the form's instruction wording."""
    27	    return hashlib.blake2b((forms.SYSTEM + "\x00" + forms.FORMS[form]).encode(),
    28	                           digest_size=8).hexdigest()

    31	def judge_sample(queries, k):
    32	    """A deterministic spread across the seeds, not the first k.
    33	
    34	    `queries` arrives grouped by seed and the seeds are score-sorted, so `queries[:50]` is the
    35	    ~10 strongest seeds' output and would overstate the on-form rate (Codex + Fable 2026-09-04).
    36	    Round-robin over seeds instead: one query from each seed before a second from any.
    37	    """
    38	    by_seed = {}
    39	    for q in queries:
    40	        by_seed.setdefault(q["seed_id"], []).append(q["query"])
    41	    out, r = [], 0
    42	    while len(out) < k and any(len(v) > r for v in by_seed.values()):
    43	        for sid in by_seed:                       # dict order == first-seen order, deterministic
    44	            if len(by_seed[sid]) > r and len(out) < k:
    45	                out.append(by_seed[sid][r])
    46	        r += 1
    47	    return out

    50	def onform_rate(verdicts, need):
    51	    """-> (rate, reason). A form can only PASS on `need` explicit boolean verdicts.
    52	
    53	    `write_md` used to print PASS from a caller-supplied float with no verdicts behind it, so
    54	    `{"onform": 0.8, "verdicts": []}` auto-approved a form without decision 15's evidence.
    55	    """
    56	    if not isinstance(verdicts, list) or len(verdicts) < need:
    57	        return None, f"only {len(verdicts) if isinstance(verdicts, list) else 0}/{need} verdicts"
    58	    vals = [v.get("on_form") if isinstance(v, dict) else v for v in verdicts[:need]]
    59	    if any(not isinstance(v, bool) for v in vals):
    60	        return None, "a verdict is not a boolean"
    61	    return sum(vals) / need, None

    64	def run(which=None, base=None, judge_n=JUDGE_N):
    65	    which = which or GENERATED
    66	    OUT.mkdir(parents=True, exist_ok=True)
    67	    kw = {"base": base} if base else {}
    68	    print(gen.health(**({"base": base} if base else {})), flush=True)
    69	    seen = set()                       # corpus-wide exact dedup across every form
    70	    seed_map, seed_meta = seedmod.cached(which, per_form=N_SEEDS)
    71	    rows = {}
    72	    for form in which:
    73	        s = seed_map[form]
    74	        if len(s) < N_SEEDS:      # a short draw is recorded, not a crash that loses every form
    75	            print(f"  WARNING {form}: only {len(s)} seeds of {N_SEEDS}", flush=True)
    76	        r = gen.generate(form, s, n=N_PER_SEED, label=form, seen=seen, **kw)
    77	        r["prompt_hash"] = prompt_hash(form)
    78	        r["contract_pass"] = r["contract_rate"] >= CONTRACT_GATE
    79	        r["judge_sample"] = judge_sample(r["queries"], judge_n)
    80	        (OUT / f"{form}.json").write_text(json.dumps(r, indent=1))
    81	        rows[form] = r
    82	    (OUT / "seed_meta.json").write_text(json.dumps(seed_meta, indent=1))
    83	    return rows, seed_meta

    86	def write_md(rows, seed_meta, path=REPO / "m10" / "SMOKE.md", judged=None):
    87	    """`judged`: {form: {"onform": float, "verdicts": [...], "note": str}} once Fable has read."""
    88	    judged = judged or {}
    89	    # Recompute every on-form rate from the verdicts themselves.
    90	    rates, reasons = {}, {}
    91	    for f in rows:
    92	        want = min(JUDGE_N, len(rows[f]["judge_sample"]))
    93	        rates[f], reasons[f] = onform_rate(judged.get(f, {}).get("verdicts", []), want)
    94	    L = []
    95	    A = L.append
    96	    A("# M10 generation smoke — decision 15 (conditional pre-approval)\n")
    97	    A(f"Generator `{gen.MODEL}` rev `{gen.REVISION[:12]}…`, vLLM on the box, thinking disabled, "
    98	      f"temperature {gen.TEMPERATURE} / top-p {gen.TOP_P}, per-request "
    99	      f"`seed = blake2b-64(seed_passage_id)`. {N_SEEDS} seed passages × {N_PER_SEED} queries = "
   100	      f"200 per form. Seeds: `{seed_meta['store']}` (Wikipedia), length "
   101	      f"{seedmod.MIN_WORDS}–{seedmod.MAX_WORDS} words, topical forms scored not first-fit; "
   102	      f"{seed_meta['dropped_protected']} of {seed_meta['screened']} dropped on the protected "
   103	      f"query index. Generated {time.strftime('%Y-%m-%d %H:%M %Z')}.\n")
   104	    A("**Disclosed bias in this sample.** For the three topical forms (`health`, `finance`, "
   105	      "`howto`) the seeds are the highest keyword-scoring passages of 400,000 Wikipedia "
   106	      "candidates, so they carry their topic more strongly than a build-scale draw would. That "
   107	      "is deliberate — this gate asks whether a PROMPT is well worded, and a health prompt on a "
   108	      "sports biography fails for a reason unrelated to wording — but a PASS here does not "
   109	      "predict the build's on-form rate. The build's own on-form rate is a separate reported "
   110	      "diagnostic with no action attached (Fable consultation, `m10/LEDGER.md` §3).\n")
   111	    A(f"**Gates.** contract ≥ {CONTRACT_GATE:.0%} (computed here) **and** on-form ≥ "
   112	      f"{ONFORM_GATE:.0%} on {JUDGE_N} queries judged by an independent Fable subagent against the "
   113	      "form's registered description. A form passing both is **auto-approved 6 h after this file "
   114	      "is pushed** unless `Dylancouzon` comments `redraft: <form>: <note>` on the GitHub issue "
   115	      "\"M10 smoke approval\" or replies via Remote Control; `approved: <form>` ends the window "
   116	      "early. **Approval binds to the prompt hash below** — a revised prompt needs a new window "
   117	      "(≤ 2 revisions, then the form is dropped from the build and its quota is not "
   118	      "redistributed).\n")
   119	    A("| form | contract | on-form | queries | out tok/s | prompt hash | verdict |")
   120	    A("|---|---|---|---|---|---|---|")
   121	    for f, r in rows.items():
   122	        of = rates[f]
   123	        ofs = "—" if of is None else f"{of:.0%}"
   124	        v = ("**PASS**" if r["contract_pass"] and of is not None and of >= ONFORM_GATE
   125	             else "**FAIL**" if of is not None or not r["contract_pass"]
   126	             else f"pending judge ({reasons[f]})")
   127	        A(f"| `{f}` | {r['contract_rate']:.0%} | {ofs} | {r['n_queries']} | "
   128	          f"{r['out_tok_per_s']:.0f} | `{r['prompt_hash']}` | {v} |")
   129	    A("")
   130	    for f, r in rows.items():
   131	        A(f"\n## `{f}`\n")
   132	        A(f"*Registered description:* {forms.FORMS[f].format(n='N')}\n")
   133	        if judged.get(f, {}).get("note"):
   134	            A(f"*Judge:* {judged[f]['note']}\n")
   135	        if r.get("transport_failures"):
   136	            A(f"**{r['transport_failures']} transport failures** — not prompt evidence.\n")
   137	        if r.get("truncated"):
   138	            A(f"{r['truncated']} replies hit the token budget (`finish_reason=length`); "
   139	              "truncation is a budget question, not a wording question.\n")
   140	        A(f"Contract {r['contract_rate']:.0%} ({r['contract_ok']}/{r['n_seeds']} seeds, "
   141	          f"{r['retried']} retried), {r['n_queries']} unique queries, "
   142	          f"{r['exact_dupes']} exact duplicates removed.\n")
   143	        if r["first_failures"]:
   144	            A(f"First contract failure, truncated: `{r['first_failures'][0][:160]}`\n")
   145	        vs = judged.get(f, {}).get("verdicts") or []
   146	        A(f"Judge sample ({len(r['judge_sample'])} of {r['n_queries']}, spread across all "
   147	          f"{r['n_seeds']} seeds){' — ✓/✗ is the judge verdict' if vs else ''}:\n")
   148	        for i, q in enumerate(r["judge_sample"]):
   149	            mark = ""
   150	            if i < len(vs):
   151	                v = vs[i].get("on_form") if isinstance(vs[i], dict) else vs[i]
   152	                mark = "✓ " if v else "✗ "
   153	            body = q if len(q) < 400 else q[:400] + "…"
   154	            A(f"- {mark}{body}")
   155	        A("")
   156	    path.write_text("\n".join(L) + "\n")
   157	    return path

    42	def judge_sample(queries, k):
    43	    """A deterministic spread across the seeds, not the first k.
    44	
    45	    `queries` arrives grouped by seed and the seeds are score-sorted, so `queries[:50]` is the
    46	    ~10 strongest seeds' output and would overstate the on-form rate (Codex + Fable 2026-09-04).
    47	    Round-robin over seeds instead: one query from each seed before a second from any.
    48	    """
    49	    by_seed = {}
    50	    for q in queries:
    51	        by_seed.setdefault(q["seed_id"], []).append(q["query"])
    52	    out, r = [], 0
    53	    while len(out) < k and any(len(v) > r for v in by_seed.values()):
    54	        for sid in by_seed:                       # dict order == first-seen order, deterministic
    55	            if len(by_seed[sid]) > r and len(out) < k:
    56	                out.append(by_seed[sid][r])
    57	        r += 1
    58	    return out
